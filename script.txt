-- Helper gọi Remote an toàn (tự thử cả Invoke và Event)
local function callRemote(name, ...)
	local ok, res = pcall(R.Invoke, name, ...)
	if ok then return true, res end
	local okEv, ev = pcall(R.Event, name)
	if okEv and ev and ev.FireServer then
		local args = { ... }
		local okFire = pcall(function() ev:FireServer(table.unpack(args)) end)
		return okFire, nil
	end
	return false, res
end

local function isOwnGui(obj)
	if guiRef and obj:IsDescendantOf(guiRef) then return true end
	local sg = obj:FindFirstAncestorOfClass("ScreenGui")
	return sg ~= nil and (sg.Name == "Game06ModernUI" or sg.Name == "Game06ToggleUI")
end

local function isVisibleOnScreen(obj)
	if not obj or not obj:IsA("GuiObject") then return false end
	if obj.AbsoluteSize.X < 10 or obj.AbsoluteSize.Y < 10 then return false end

	local sg = obj:FindFirstAncestorOfClass("ScreenGui")
	if not sg or not sg.Enabled then return false end

	local cur = obj
	while cur and cur:IsA("GuiObject") do
		if not cur.Visible then return false end
		cur = cur.Parent
	end

	local cam = workspace.CurrentCamera
	if cam then
		local vp = cam.ViewportSize
		local pos = obj.AbsolutePosition
		local sz = obj.AbsoluteSize
		if pos.X + sz.X < 0 or pos.Y + sz.Y < 0 or pos.X > vp.X or pos.Y > vp.Y then
			return false
		end
	end
	return true
end

local function waitFeature(sec, feature)
	local t = 0
	while t < sec and isOn(feature) do
		local dt = task.wait(math.min(0.2, sec - t))
		t = t + dt
	end
	return isOn(feature)
end

local function waitAnyThrow(sec)
	local t = 0
	while t < sec and (isOn("autoThrow") or isOn("autoCollectIndex")) do
		local dt = task.wait(math.min(0.2, sec - t))
		t = t + dt
	end
	return isOn("autoThrow") or isOn("autoCollectIndex")
end

local function getDiscoveredSet()
	local data = getPlayerData()
	local set = {}

	local function collectNames(tbl, depth)
		if type(tbl) ~= "table" or depth > 4 then return end
		for k, v in pairs(tbl) do
			if type(k) == "number" and type(v) == "string" then
				set[v] = true
				set[string.lower(v)] = true
			elseif type(k) == "string" then
				if v == true or (type(v) == "number" and v > 0) then
					set[k] = true
					set[string.lower(k)] = true
				elseif type(v) == "table" then
					-- Nếu bảng con có dạng { Normal = true, Gold = true }
					local isEntry = false
					for subK, subV in pairs(v) do
						if subV == true or (type(subV) == "number" and subV > 0) then
							isEntry = true
							if type(subK) == "string" then
								set[k .. ":" .. subK] = true
							end
						end
					end
					if isEntry then
						set[k] = true
						set[string.lower(k)] = true
					end
					collectNames(v, depth + 1)
				end
			end
		end
	end

	for k, v in pairs(data) do
		local lk = string.lower(tostring(k))
		if type(v) == "table" and not lk:find("claim") and not lk:find("reward") then
			if lk:find("index") or lk:find("discover") or lk:find("seen") or lk:find("catalog") or lk:find("unlock") and lk:find("animal") then
				collectNames(v, 1)
			end
		end
	end

	return set
end

local function getClaimedSet()
	local data = getPlayerData()
	local claimed = {}

	for k, v in pairs(data) do
		local lk = string.lower(tostring(k))
		if type(v) == "table" and (lk:find("claim") or (lk:find("index") and lk:find("reward"))) then
			for ck, cv in pairs(v) do
				if type(ck) == "number" and type(cv) == "string" then
					claimed[cv] = true
				elseif cv == true or (type(cv) == "number" and cv > 0) then
					claimed[tostring(ck)] = true
				end
			end
		end
	end

	return claimed
end

	owned[eq] = true
	owned[1] = true

	for k, v in pairs(data) do
		local lk = string.lower(tostring(k))
		if lk:find("food") and lk ~= "equippedfood" then
			if type(v) == "number" then
				for i = 1, v do owned[i] = true end
			elseif type(v) == "table" then
				for fk, fv in pairs(v) do
					if type(fk) == "number" and type(fv) == "number" then
						owned[fv] = true
					elseif fv == true or (type(fv) == "number" and fv > 0) then
						local idx = tonumber(fk)
						if idx then owned[idx] = true end
					end
				end
			end
		end
	end

	-- Dự phòng: mọi thức ăn <= EquippedFood thường đều đã sở hữu
	if type(eq) == "number" then
		for i = 1, eq do owned[i] = true end
	end

	return owned
end

local function getOrderedAreas()
	local list = {}
	if not AreasConfig then return list end

	local src = AreasConfig.List or AreasConfig.All
	if type(src) == "table" then
		for k, v in pairs(src) do
			if type(v) == "table" then
				local key = v.Key or (type(k) == "string" and k) or v.Name
				local playable = true
				if type(AreasConfig.IsPlayable) == "function" and key then
					local ok, p = pcall(AreasConfig.IsPlayable, key)
					if ok and p == false then playable = false end
				end
				if playable then
					table.insert(list, v)
				end
			end
		end
	end

	table.sort(list, function(a, b)
		local da = a.StartDistance or a.Index or 0
		local db = b.StartDistance or b.Index or 0
		return da < db
	end)
	return list
end

local function getAreaAnimals(area)
	local animals = {}
	if not AnimalEconomy then return animals end

	local key = area.Key or area.Name
	local got = nil
	if type(AnimalEconomy.GetAreaAnimals) == "function" then
		local ok, res = pcall(AnimalEconomy.GetAreaAnimals, key)
		if ok and type(res) == "table" and next(res) ~= nil then
			got = res
		end
	end

	if got then
		for k, v in pairs(got) do
			if type(v) == "string" then
				local info = type(AnimalEconomy.Get) == "function" and AnimalEconomy.Get(v) or {}
				table.insert(animals, { Name = v, Rarity = info and info.Rarity, Rank = info and info.Rank or 0, Info = info })
			elseif type(v) == "table" then
				local name = v.Name or v.Key or v.Id or (type(k) == "string" and k)
				if name then
					table.insert(animals, { Name = name, Rarity = v.Rarity, Rank = v.Rank or 0, Info = v })
				end
			end
		end
	elseif type(AnimalEconomy.List) == "table" then
		for k, v in pairs(AnimalEconomy.List) do
			if type(v) == "table" and (v.Area == key or v.Area == area.Name or v.Area == area.Index) then
				local name = v.Name or v.Key or v.Id or tostring(k)
				table.insert(animals, { Name = name, Rarity = v.Rarity, Rank = v.Rank or 0, Info = v })
			end
		end
	end

	table.sort(animals, function(a, b)
		return (a.Rank or 0) < (b.Rank or 0)
	end)
	return animals
end

-- Đánh giá thức ăn sở hữu nào phù hợp với pet còn thiếu (dựa trên Bait / Rarity / Tier)
local function scoreFoodForAnimal(food, animal, area)
	if type(food) ~= "table" then return 0 end
	local rarity = animal.Rarity and tostring(animal.Rarity) or ""
	local lRarity = string.lower(rarity)
	local lName = string.lower(tostring(animal.Name or ""))
	local lArea = string.lower(tostring(area.Key or area.Name or ""))
	local bait = food.Bait

	if type(bait) == "table" then
		-- Dạng bảng trọng số { Common = 50, Rare = 30... } hoặc danh sách
		local direct = bait[rarity] or bait[animal.Rarity] or bait[animal.Name]
		if type(direct) == "number" then
			return direct > 0 and (1000 + direct) or 0
		elseif direct == true then
			return 1500
		end
		local hasStringList = false
		for bk, bv in pairs(bait) do
			if type(bk) == "number" and type(bv) == "string" then
				hasStringList = true
				local lb = string.lower(bv)
				if lb == lName or lb == lRarity or lb == lArea then
					return 1500
				end
			elseif type(bk) == "string" and (string.lower(bk) == lRarity or string.lower(bk) == lName) then
				if type(bv) == "number" then return bv > 0 and (1000 + bv) or 0 end
				if bv == true then return 1500 end
			end
		end
		-- Nếu Bait là bảng lọc nhưng không có mục phù hợp với pet này
		if hasStringList or next(bait) ~= nil then
			return 0
		end
	elseif type(bait) == "string" and bait ~= "" then
		local lb = string.lower(bait)
		if lb == lName or lb == lRarity or lb == lArea or lb == "all" or lb == "any" then
			return 1500
		end
		return 0
	end

	-- Dự phòng theo Rarity / Tier của thức ăn
	if food.Rarity and lRarity ~= "" and string.lower(tostring(food.Rarity)) == lRarity then
		return 800 + (food.Index or 0)
	end
	return 100 + (food.Tier or food.Index or 0)
end

local function findBestOwnedFood(animal, area, ownedFoods)
	local bestIdx, bestScore = nil, 0
	for foodIdx in pairs(ownedFoods) do
		local food = Foods.Get and Foods.Get(foodIdx) or (Foods.GetAny and Foods.GetAny(foodIdx))
		if food then
			local sc = scoreFoodForAnimal(food, animal, area)
			if sc > bestScore then
				bestScore = sc
				bestIdx = foodIdx
			end
		end
	end
	return bestIdx, bestScore
end

-- Ước lượng tầm ném tối đa của nhân vật
local observedMaxDistance = nil
local function getEstimatedMaxDistance()
	if observedMaxDistance and observedMaxDistance > 0 then
		return observedMaxDistance
	end
	local data = getPlayerData()
	for _, k in ipairs({ "MaxDistance", "BestDistance", "FurthestDistance", "ThrowDistance", "MaxThrowDistance" }) do
		local v = data[k] or P:GetAttribute(k)
		if type(v) == "number" and v > 0 then
			return v
		end
	end
	-- Thử tính từ LevelConfig nếu có hàm
	if LevelConfig then
		local lvl = data.Level or D:Get("Level") or P:GetAttribute("Level")
		if type(lvl) == "number" then
			for _, fnName in ipairs({ "GetMaxDistance", "GetThrowDistance", "DistanceForLevel", "GetDistance" }) do
				if type(LevelConfig[fnName]) == "function" then
					local ok, d = pcall(LevelConfig[fnName], lvl)
					if ok and type(d) == "number" and d > 0 then
						return d
					end
				end
			end
		end
	end
	return nil
end

-- Đặt lại lực ném qua Player_SetThrowPower đúng đơn vị của game (0..1 hoặc 1..100)
local originalPowerValue = nil
local function setThrowPowerRatio(ratio)
	local data = getPlayerData()
	local cur = data.ThrowPower or data.ThrowPowerPercent or P:GetAttribute("ThrowPower")
	if originalPowerValue == nil and type(cur) == "number" then
		originalPowerValue = cur
	end
	local isPercent = type(cur) == "number" and cur > 1.5
	local value = isPercent and math.clamp(math.floor(ratio * 100 + 0.5), 1, 100) or math.clamp(ratio, 0.05, 1)
	callRemote("Player_SetThrowPower", value)
end

local function restoreThrowPower()
	if originalPowerValue ~= nil then
		callRemote("Player_SetThrowPower", originalPowerValue)
		originalPowerValue = nil
	end
end

-- Tìm mục tiêu Index tiếp theo
local activeIndexPlan = nil

local function computeIndexPlan()
	local areas = getOrderedAreas()
	if #areas == 0 then
		return nil, "Không đọc được AreasConfig"
	end

	local discovered = getDiscoveredSet()
	local ownedFoods = getOwnedFoods()
	local maxDist = getEstimatedMaxDistance()
	local blockedAreaName = nil

	for _, area in ipairs(areas) do
		local areaName = area.Name or area.Key or "Unknown"
		local startDist = area.StartDistance or 0
		local targetDist = startDist + (area.Length or 0) * 0.45
		local animals = getAreaAnimals(area)

		for _, animal in ipairs(animals) do
			local hasIt = discovered[animal.Name] or discovered[string.lower(animal.Name)]
			if not hasIt then
				local powerEnough = (maxDist == nil) or (maxDist >= startDist)
				local foodIdx = findBestOwnedFood(animal, area, ownedFoods)

				if powerEnough and foodIdx then
					local ratio = 1
					if maxDist and maxDist > 0 then
						ratio = math.clamp(targetDist / maxDist, 0.08, 1)
					end
					return {
						Area = area,
						AreaName = areaName,
						Animal = animal,
						FoodIndex = foodIdx,
						PowerRatio = ratio,
					}, "Collecting " .. animal.Name .. " (" .. areaName .. ")"
				else
					if not blockedAreaName then
						blockedAreaName = areaName
					end
				end
			end
		end
	end

	if blockedAreaName then
		return nil, "Missing pets in " .. blockedAreaName .. " need more power or a food you own"
	end
	return nil, "All index pets collected!"
end

local function isTraining(c, hrp)
	if not c or not hrp or not hrp.Parent then return false end
	if hrp.Anchored then return true end
	if P:GetAttribute("IsTraining") == true or P:GetAttribute("Training") == true then
		return true
	end
	for _, obj in ipairs(playerGui:GetDescendants()) do
		if not isOwnGui(obj) and (obj:IsA("TextButton") or obj:IsA("TextLabel")) then
			local raw = string.gsub(obj.Text or "", "<[^>]->", "")
			local clean = string.gsub(string.lower(raw), "%s+", "")
			if (clean == "dừng" or clean == "dung" or clean == "stop") and isVisibleOnScreen(obj) then
				return true
			end
		end
	end
	return false
end

local stepOffsets = {
	Vector3.new(0, 2.5, 5),
	Vector3.new(5, 2.5, 0),
	Vector3.new(-5, 2.5, 0),
	Vector3.new(0, 2.5, -5),
}

local function enterTraining(myPlot, c, h, hrp, maxTries)
	local spot = myPlot.Position + CFG.Plot.TrainingSpotOffset
	maxTries = maxTries or 4

	for attempt = 1, maxTries do
		if not (isOn("autoThrow") or isOn("autoCollectIndex")) or not hrp.Parent or h.Health <= 0 then
			return false
		end
		if isTraining(c, hrp) then
			return true
		end

		pcall(function()
			hrp.AssemblyLinearVelocity = Vector3.zero
			hrp.AssemblyAngularVelocity = Vector3.zero
		end)

		if attempt == 1 then
			hrp.CFrame = CFrame.new(spot + Vector3.new(0, 2, 0))
			h:MoveTo(spot)
			callRemote("Training_Start")
		else
			local off = stepOffsets[((attempt - 2) % #stepOffsets) + 1]
			hrp.CFrame = CFrame.new(spot + off, spot)
			if not waitAnyThrow(0.25) then return false end
			h:MoveTo(spot)
			if not waitAnyThrow(0.45) then return false end
			if not isTraining(c, hrp) then
				hrp.CFrame = CFrame.new(spot + Vector3.new(0, 2, 0))
				h:MoveTo(spot)
				callRemote("Training_Start")
			end
		end

		for _ = 1, 6 do
			if not (isOn("autoThrow") or isOn("autoCollectIndex")) then return false end
			if isTraining(c, hrp) then return true end
			task.wait(0.1)
		end
	end

	return isTraining(c, hrp)
end

local function waitReady(myPlot, c, h, hrp)
	local lastCheck = os.clock()
	while isOn("autoThrow") or isOn("autoCollectIndex") do
		if not c.Parent or not hrp.Parent or h.Health <= 0 then
			return false
		end
		local at = P:GetAttribute("ThrowReadyAt")
		if not at or at - workspace:GetServerTimeNow() <= 0.3 then
			return true
		end
		if os.clock() - lastCheck >= 1.2 then
			lastCheck = os.clock()
			if not isTraining(c, hrp) then
				enterTraining(myPlot, c, h, hrp, 2)
			end
		end
		task.wait(0.1)
	end
	return false
end

			if D:Get("EquippedFood") ~= plan.FoodIndex then

			end

			local rs = R.Invoke("Throw_Resolve", 1)
			if type(rs) == "table" and rs.Ok then
				local resolvedDist = rs.Distance or rs.MaxDistance
				if type(resolvedDist) == "number" and resolvedDist > 0 and (not plan or plan.PowerRatio >= 0.99) then
					observedMaxDistance = math.max(observedMaxDistance or 0, resolvedDist)
				end

				if not waitAnyThrow(0.4) then continue end
				h.Health = 0
				local newChar = P.CharacterAdded:Wait()
				newChar:WaitForChild("HumanoidRootPart", 5)
				waitAnyThrow(0.6)
			end
		else
			waitAnyThrow(0.5)
		end
	end
end)

local lastTapAt = setmetatable({}, { __mode = "k" })

local function isX2Popup(obj)
	if isOwnGui(obj) then return false end
	if obj:IsA("TextLabel") or obj:IsA("TextButton") then
		local raw = string.gsub(obj.Text or "", "<[^>]->", "")
		local clean = string.gsub(string.lower(raw), "%s+", "")
		if clean == "x2" or clean == "2x" then
			return true
		end
	end
	local lowerName = string.lower(obj.Name)
	if obj:IsA("GuiButton") and (string.find(lowerName, "bonus") or string.find(lowerName, "tapx2")) then
		return true
	end
	return false
end

local function findClickableTarget(obj)
	local cur = obj
	for _ = 1, 5 do
		if not cur or cur:IsA("ScreenGui") then break end
		if cur:IsA("GuiButton") then return cur end
		cur = cur.Parent
	end
	local parent = obj.Parent
	if parent and parent:IsA("GuiObject") and parent.AbsoluteSize.X <= 300 and parent.AbsoluteSize.Y <= 300 then
		return parent
	end
	return obj
end

local function tapTarget(target)
	if not isVisibleOnScreen(target) then return end
	local now = os.clock()
	if lastTapAt[target] and now - lastTapAt[target] < 0.18 then return end
	lastTapAt[target] = now

	local firedBySignal = false
	if target:IsA("GuiButton") and type(getconnections) == "function" then
		for _, sig in ipairs({ target.Activated, target.MouseButton1Down, target.MouseButton1Click }) do
			pcall(function()
				for _, conn in ipairs(getconnections(sig)) do
					if conn.Fire then
						conn:Fire()
						firedBySignal = true
					elseif conn.Function then
						task.spawn(conn.Function)
						firedBySignal = true
					end
				end
			end)
		end
	end

	if not firedBySignal and VIM then
		local sg = target:FindFirstAncestorOfClass("ScreenGui")
		local inset = (sg and sg.IgnoreGuiInset) and Vector2.zero or GuiService:GetGuiInset()
		local center = target.AbsolutePosition + target.AbsoluteSize / 2 + inset
		pcall(function()
			VIM:SendMouseButtonEvent(center.X, center.Y, 0, true, game, 1)
			task.wait(0.03)
			VIM:SendMouseButtonEvent(center.X, center.Y, 0, false, game, 1)
		end)
	end
end
